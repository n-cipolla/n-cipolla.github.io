Interactive Fourier Transform Visualization

import numpy as np import matplotlib.pyplot as plt from pyscript import Element plt.rcParams["font.size"] = 12 def create_wave(A,f,t): return A*np.sin(2*np.pi*f*t) def update(event=None): f1=float(Element("f1").value) f2=float(Element("f2").value) f3=float(Element("f3").value) N=4000 dt=1/800 t=np.arange(0,N)*dt y=create_wave(1,f1,t)+create_wave(1,f2,t)+create_wave(1,f3,t) freqs=np.fft.rfftfreq(N,dt) spectrum=np.abs(np.fft.rfft(y)) fig,(ax1,ax2)=plt.subplots(1,2,figsize=(10,4)) ax1.plot(t,y) ax1.set_xlim(0,2) ax1.set_title("Time Domain") ax2.plot(freqs,spectrum) ax2.set_xlim(0,25) ax2.set_title("Fourier Transform") Element("plot").write(fig) update()
